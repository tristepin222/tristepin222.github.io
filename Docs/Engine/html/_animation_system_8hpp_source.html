134 void updateController(AnimationControllerComponent& controller, AnimatorComponent& animator, float dt) {
135 if ((controller.currentState.empty() || controller.currentState == "Entry" || controller.currentState == "__Entry__") && !controller.states.empty()) {
136 controller.currentState = controller.states[0].name;
137 controller.currentStateTime = 0.0f;
138 }
139
140 if (controller.currentState.empty()) return;
141
142 // Advance play time for current state
143 const AnimationState* currState = nullptr;
144 for (const auto& state : controller.states) {
145 if (state.name == controller.currentState) {
146 currState = &state;
147 break;
148 }
149 }
150
151 if (currState) {
152 controller.currentStateTime += dt * currState->speed;
153
154 // Sync active clip & time in AnimatorComponent from current state
155 if (!currState->isBlendTree) {
156 int clipIdx = -1;
157 if (!currState->clipName.empty()) {
158 for (size_t i = 0; i < animator.animations.size(); ++i) {
159 if (animator.animations[i].name == currState->clipName) {
160 clipIdx = static_cast<int>(i);
161 break;
162 }
163 }
164 }
165 if (clipIdx == -1 && !animator.animations.empty()) {
166 clipIdx = 0;
167 }
168 if (clipIdx != -1) {
169 animator.activeAnimationIndex = clipIdx;
170 auto& clip = animator.animations[clipIdx];
171
172 float effectiveDur = clip.duration;
173 for (const auto& chan : clip.propertyChannels) {
174 if (!chan.keys.empty()) {
175 effectiveDur = std::max(effectiveDur, chan.keys.back().time);
176 }
177 }
178 if (effectiveDur <= 0.0f) effectiveDur = 1.0f;
179 clip.duration = effectiveDur;
180
181 if (currState->isLooping) {
182 animator.currentTime = std::fmod(controller.currentStateTime, effectiveDur);
183 } else {
184 animator.currentTime = std::min(controller.currentStateTime, effectiveDur);
185 }
186 animator.loop = currState->isLooping;
187 }
188 }
189 }
190
191 // Advance crossfade timing if active
192 if (controller.isCrossfading) {
193 const AnimationState* fState = nullptr;
194 for (const auto& state : controller.states) {
195 if (state.name == controller.fromState) {
196 fState = &state;
197 break;
198 }
199 }
200 if (fState) {
201 controller.fromStateTime += dt * fState->speed;
202 }
203
204 controller.crossfadeProgress += dt;
205 if (controller.crossfadeProgress >= controller.crossfadeDuration) {
206 controller.isCrossfading = false;
207 controller.fromState.clear();
208 }
209 }
210
211 // Check transition rules (only if not currently crossfading)
212 if (!controller.isCrossfading) {
213 for (const auto& trans : controller.transitions) {
214 if (trans.fromState == controller.currentState) {
215 bool allConditionsMet = true;
216 for (const auto& cond : trans.conditions) {
217 float paramVal = 0.0f;
218 auto it = controller.parameters.find(cond.parameterName);
219 if (it != controller.parameters.end()) {
220 paramVal = it->second;
221 }
222
223 if (cond.op == ">") {
224 if (!(paramVal > cond.value)) allConditionsMet = false;
225 } else if (cond.op == "<") {
226 if (!(paramVal < cond.value)) allConditionsMet = false;
227 } else if (cond.op == "==") {
228 if (!(std::abs(paramVal - cond.value) < 1e-4f)) allConditionsMet = false;
229 } else {
230 allConditionsMet = false;
231 }
232 }
233
234 bool shouldTransition = false;
235 if (!trans.conditions.empty()) {
236 shouldTransition = allConditionsMet;
237 } else {
238 float clipDur = 0.5f;
239 if (currState) {
240 for (const auto& c : animator.animations) {
241 if (c.name == currState->clipName) {
242 clipDur = c.duration;
243 break;
244 }
245 }
246 }
247 if (clipDur <= 0.0f || controller.currentStateTime >= clipDur) {
248 shouldTransition = true;
249 }
250 }
251
252 if (shouldTransition) {
253 // Trigger transition!
254 controller.fromState = controller.currentState;
255 controller.fromStateTime = controller.currentStateTime;
256 controller.currentState = trans.toState;
257 controller.currentStateTime = 0.0f;
258
259 controller.targetState = trans.toState;
260 controller.crossfadeProgress = 0.0f;
261 controller.crossfadeDuration = trans.crossfadeDuration;
262 controller.isCrossfading = true;
263 break; // only transition once per tick
264 }
265 }
266 }
267 }

297 void evaluateStatePose(const AnimationControllerComponent& controller, const AnimationState& state, float stateTime, const AnimatorComponent& animator, const SkeletonComponent& skeleton, std::vector<JointPose>& outPose) {
298 if (state.isBlendTree) {
299 const auto& tree = state.blendTree;
300 if (tree.nodes.empty()) return;
301
302 float paramVal = 0.0f;
303 auto it = controller.parameters.find(tree.parameterName);
304 if (it != controller.parameters.end()) {
305 paramVal = it->second;
306 }
307
308 if (tree.is2D) {
309 float paramX = 0.0f;
310 auto itX = controller.parameters.find(tree.parameterName);
311 if (itX != controller.parameters.end()) {
312 paramX = itX->second;
313 }
314
315 float paramY = 0.0f;
316 auto itY = controller.parameters.find(tree.parameterYName);
317 if (itY != controller.parameters.end()) {
318 paramY = itY->second;
319 }
320
321 glm::vec2 p(paramX, paramY);
322 size_t numNodes = tree.nodes.size();
323 std::vector<float> weights(numNodes, 0.0f);
324 float totalWeight = 0.0f;
325
326 // 2D Freeform Cartesian / Gradient Band Blending algorithm
327 for (size_t i = 0; i < numNodes; ++i) {
328 glm::vec2 pi = tree.nodes[i].threshold2D;
329 float minVal = 1.0f;
330
331 for (size_t j = 0; j < numNodes; ++j) {
332 if (i == j) continue;
333 glm::vec2 pj = tree.nodes[j].threshold2D;
334 glm::vec2 v = pj - pi;
335 float lenSq = glm::dot(v, v);
336 if (lenSq < 1e-5f) continue;
337
338 glm::vec2 u = p - pi;
339 float t = glm::dot(u, v) / lenSq;
340 float val = 1.0f - t;
341 if (val < minVal) {
342 minVal = val;
343 }
344 }
345
346 weights[i] = std::max(0.0f, minVal);
347 totalWeight += weights[i];
348 }
349
350 // Normalize weights and sample/blend poses
351 if (totalWeight > 1e-5f) {
352 for (size_t i = 0; i < numNodes; ++i) {
353 weights[i] /= totalWeight;
354 }
355
356 std::vector<JointPose> blendedPose(skeleton.joints.size());
357 for (size_t j = 0; j < skeleton.joints.size(); ++j) {
358 blendedPose[j] = { skeleton.joints[j].bindTranslation, skeleton.joints[j].bindRotation, skeleton.joints[j].bindScale };
359 }
360
361 float cumulativeWeight = 0.0f;
362 bool firstPoseSet = false;
363
364 for (size_t i = 0; i < numNodes; ++i) {
365 float w = weights[i];
366 if (w < 1e-4f) continue;
367
368 if (const AnimationClip* clip = findClip(animator, tree.nodes[i].clipName)) {
369 std::vector<JointPose> tempPose(skeleton.joints.size());
370 for (size_t j = 0; j < skeleton.joints.size(); ++j) {
371 tempPose[j] = { skeleton.joints[j].bindTranslation, skeleton.joints[j].bindRotation, skeleton.joints[j].bindScale };
372 }
373 sampleClip(*clip, stateTime, skeleton, tempPose);
374
375 if (!firstPoseSet) {
376 blendedPose = tempPose;
377 cumulativeWeight = w;
378 firstPoseSet = true;
379 } else {
380 float blendFactor = w / (cumulativeWeight + w);
381 blendPoses(blendedPose, tempPose, blendFactor, blendedPose);
382 cumulativeWeight += w;
383 }
384 }
385 }
386 outPose = blendedPose;
387 }
388 } else {
389 // Sort nodes by threshold for robust linear 1D interpolation
390 std::vector<BlendNode> sortedNodes = tree.nodes;
391 std::sort(sortedNodes.begin(), sortedNodes.end(), [](const BlendNode& a, const BlendNode& b) {
392 return a.threshold < b.threshold;
393 });
394
395 if (paramVal <= sortedNodes.front().threshold) {
396 if (const AnimationClip* clip = findClip(animator, sortedNodes.front().clipName)) {
397 sampleClip(*clip, stateTime, skeleton, outPose);
398 }
399 } else if (paramVal >= sortedNodes.back().threshold) {
400 if (const AnimationClip* clip = findClip(animator, sortedNodes.back().clipName)) {
401 sampleClip(*clip, stateTime, skeleton, outPose);
402 }
403 } else {
404 for (size_t i = 0; i < sortedNodes.size() - 1; ++i) {
405 if (paramVal >= sortedNodes[i].threshold && paramVal <= sortedNodes[i+1].threshold) {
406 float t = (paramVal - sortedNodes[i].threshold) / (sortedNodes[i+1].threshold - sortedNodes[i].threshold);
407
408 std::vector<JointPose> poseA(skeleton.joints.size());
409 std::vector<JointPose> poseB(skeleton.joints.size());
410 for (size_t j = 0; j < skeleton.joints.size(); ++j) {
411 poseA[j] = { skeleton.joints[j].bindTranslation, skeleton.joints[j].bindRotation, skeleton.joints[j].bindScale };
412 poseB[j] = poseA[j];
413 }
414
415 if (const AnimationClip* clipA = findClip(animator, sortedNodes[i].clipName)) {
416 sampleClip(*clipA, stateTime, skeleton, poseA);
417 }
418 if (const AnimationClip* clipB = findClip(animator, sortedNodes[i+1].clipName)) {
419 sampleClip(*clipB, stateTime, skeleton, poseB);
420 }
421
422 blendPoses(poseA, poseB, t, outPose);
423 break;
424 }
425 }
426 }
427 }
428 } else {
429 if (const AnimationClip* clip = findClip(animator, state.clipName)) {
430 sampleClip(*clip, stateTime, skeleton, outPose);
431 }
432 }

444 glm::quat rotationBetweenVectors(glm::vec3 u, glm::vec3 v) {
445 u = glm::normalize(u);
446 v = glm::normalize(v);
447 float cosTheta = glm::dot(u, v);
448 glm::vec3 rotationAxis;
449
450 if (cosTheta < -1.0f + 1e-6f) {
451 rotationAxis = glm::cross(glm::vec3(0.0f, 0.0f, 1.0f), u);
452 if (glm::dot(rotationAxis, rotationAxis) < 0.01f) {
453 rotationAxis = glm::cross(glm::vec3(1.0f, 0.0f, 0.0f), u);
454 }
455 rotationAxis = glm::normalize(rotationAxis);
456 return glm::angleAxis(glm::radians(180.0f), rotationAxis);
457 }
458
459 rotationAxis = glm::cross(u, v);
460 float s = std::sqrt((1.0f + cosTheta) * 2.0f);
461 float invs = 1.0f / s;
462
463 return glm::quat(
464 s * 0.5f,
465 rotationAxis.x * invs,
466 rotationAxis.y * invs,
467 rotationAxis.z * invs
468 );
469 }

471 void solve2BoneIK(SkeletonComponent& skeleton, const IKSolverComponent& ik, std::vector<glm::mat4>& globalTransforms) {
472 int idxA = findJointIndex(skeleton, ik.startJointName);
473 int idxB = findJointIndex(skeleton, ik.middleJointName);
474 int idxC = findJointIndex(skeleton, ik.endJointName);
475
476 if (idxA == -1 || idxB == -1 || idxC == -1) return;
477
478 glm::mat4 origGlobalA = globalTransforms[idxA];
479 glm::mat4 origGlobalB = globalTransforms[idxB];
480 glm::mat4 origGlobalC = globalTransforms[idxC];
481
482 glm::vec3 P_A = glm::vec3(origGlobalA[3]);
483 glm::vec3 P_B = glm::vec3(origGlobalB[3]);
484 glm::vec3 P_C = glm::vec3(origGlobalC[3]);
485 glm::vec3 P_T = ik.targetPosition;
486 glm::vec3 P_P = ik.polePosition;
487
488 float L1 = glm::distance(P_B, P_A);
489 float L2 = glm::distance(P_C, P_B);
490 float D = glm::distance(P_T, P_A);
491
492 if (L1 < 1e-4f || L2 < 1e-4f) return;
493
494 if (D > L1 + L2 - 0.001f) {
495 P_T = P_A + glm::normalize(P_T - P_A) * (L1 + L2 - 0.001f);
496 D = L1 + L2 - 0.001f;
497 }
498 if (D < 0.001f) {
499 P_T = P_A + glm::vec3(0.0f, 0.0f, 0.001f);
500 D = 0.001f;
501 }
502
503 float cosAlpha = (L1 * L1 + D * D - L2 * L2) / (2.0f * L1 * D);
504 cosAlpha = glm::clamp(cosAlpha, -1.0f, 1.0f);
505 float alpha = std::acos(cosAlpha);
506
507 glm::vec3 d_AT = glm::normalize(P_T - P_A);
508 glm::vec3 v_AP = P_P - P_A;
509 glm::vec3 N = glm::cross(d_AT, v_AP);
510 if (glm::dot(N, N) < 1e-6f) {
511 N = glm::cross(d_AT, glm::vec3(0.0f, 1.0f, 0.0f));
512 if (glm::dot(N, N) < 1e-6f) {
513 N = glm::cross(d_AT, glm::vec3(1.0f, 0.0f, 0.0f));
514 }
515 }
516 N = glm::normalize(N);
517 glm::vec3 B_dir = glm::normalize(glm::cross(N, d_AT));
518
519 glm::vec3 P_B_new = P_A + (d_AT * std::cos(alpha) + B_dir * std::sin(alpha)) * L1;
520 glm::vec3 P_C_new = P_T;
521
522 glm::vec3 origDirA = P_B - P_A;
523 glm::vec3 targetDirA = P_B_new - P_A;
524 glm::quat rotA = rotationBetweenVectors(origDirA, targetDirA);
525 glm::mat3 rotPartA = glm::mat3(glm::toMat4(rotA)) * glm::mat3(origGlobalA);
526 glm::mat4 newGlobalA = glm::mat4(rotPartA);
527 newGlobalA[3] = glm::vec4(P_A, 1.0f);
528
529 glm::vec3 origDirB = P_C - P_B;
530 glm::vec3 targetDirB = P_C_new - P_B_new;
531 glm::quat rotB = rotationBetweenVectors(origDirB, targetDirB);
532 glm::mat3 rotPartB = glm::mat3(glm::toMat4(rotB)) * glm::mat3(origGlobalB);
533 glm::mat4 newGlobalB = glm::mat4(rotPartB);
534 newGlobalB[3] = glm::vec4(P_B_new, 1.0f);
535
536 glm::mat4 newGlobalC = origGlobalC;
537 newGlobalC[3] = glm::vec4(P_C_new, 1.0f);
538
539 float w = glm::clamp(ik.targetWeight, 0.0f, 1.0f);
540 globalTransforms[idxA] = origGlobalA * (1.0f - w) + newGlobalA * w;
541 globalTransforms[idxB] = origGlobalB * (1.0f - w) + newGlobalB * w;
542 globalTransforms[idxC] = origGlobalC * (1.0f - w) + newGlobalC * w;
543
544 auto updateLocal = [&](int idx) {
545 int parentIdx = skeleton.joints[idx].parentIndex;
546 if (parentIdx != -1) {
547 skeleton.joints[idx].localTransform = glm::inverse(globalTransforms[parentIdx]) * globalTransforms[idx];
548 } else {
549 skeleton.joints[idx].localTransform = globalTransforms[idx];
550 }
551 };
552
553 updateLocal(idxA);
554 updateLocal(idxB);
555 updateLocal(idxC);
556
557 std::vector<bool> resolved(skeleton.joints.size(), false);
558 resolved[idxA] = true;
559 resolved[idxB] = true;
560 resolved[idxC] = true;
561
562 std::function<void(int)> propagate = [&](int jointIdx) {
563 int parentIdx = skeleton.joints[jointIdx].parentIndex;
564 if (parentIdx != -1 && resolved[parentIdx] && !resolved[jointIdx]) {
565 globalTransforms[jointIdx] = globalTransforms[parentIdx] * skeleton.joints[jointIdx].localTransform;
566 resolved[jointIdx] = true;
567 }
568 for (size_t i = 0; i < skeleton.joints.size(); ++i) {
569 if (skeleton.joints[i].parentIndex == jointIdx) {
570 propagate(static_cast<int>(i));
571 }
572 }
573 };
574
575 propagate(idxA);
576 propagate(idxB);
577 propagate(idxC);
578 }

580 void solveFABRIK(SkeletonComponent& skeleton, const IKSolverComponent& ik, std::vector<glm::mat4>& globalTransforms) {
581 if (ik.jointChainNames.size() < 2) return;
582
583 std::vector<int> chainIndices;
584 for (const auto& name : ik.jointChainNames) {
585 int idx = findJointIndex(skeleton, name);
586 if (idx != -1) {
587 chainIndices.push_back(idx);
588 }
589 }
590
591 if (chainIndices.size() < 2) return;
592
593 int n = static_cast<int>(chainIndices.size()) - 1;
594 std::vector<glm::vec3> positions(chainIndices.size());
595 std::vector<float> boneLengths(n);
596 float totalLength = 0.0f;
597
598 for (size_t i = 0; i < chainIndices.size(); ++i) {
599 positions[i] = glm::vec3(globalTransforms[chainIndices[i]][3]);
600 if (i > 0) {
601 boneLengths[i - 1] = glm::distance(positions[i], positions[i - 1]);
602 totalLength += boneLengths[i - 1];
603 }
604 }
605
606 glm::vec3 target = ik.targetPosition;
607 glm::vec3 origin = positions[0];
608
609 float distToTarget = glm::distance(origin, target);
610 if (distToTarget > totalLength) {
611 for (int i = 0; i < n; ++i) {
612 glm::vec3 dir = glm::normalize(target - positions[i]);
613 positions[i + 1] = positions[i] + dir * boneLengths[i];
614 }
615 } else {
616 for (int iter = 0; iter < ik.maxIterations; ++iter) {
617 float err = glm::distance(positions[n], target);
618 if (err < ik.tolerance) break;
619
620 positions[n] = target;
621 for (int i = n - 1; i >= 0; --i) {
622 glm::vec3 dir = glm::normalize(positions[i] - positions[i + 1]);
623 positions[i] = positions[i + 1] + dir * boneLengths[i];
624 }
625
626 positions[0] = origin;
627 for (int i = 0; i < n; ++i) {
628 glm::vec3 dir = glm::normalize(positions[i + 1] - positions[i]);
629 positions[i + 1] = positions[i] + dir * boneLengths[i];
630 }
631 }
632 }
633
634 for (int i = 0; i < n; ++i) {
635 int idxCurrent = chainIndices[i];
636 int idxNext = chainIndices[i + 1];
637
638 glm::mat4 origGlobalCurrent = globalTransforms[idxCurrent];
639 // Compute current position of idxNext by evaluating relative to current idxCurrent transform
640 glm::mat4 currentGlobalNext = origGlobalCurrent * skeleton.joints[idxNext].localTransform;
641
642 glm::vec3 P_current = glm::vec3(origGlobalCurrent[3]);
643 glm::vec3 P_next = glm::vec3(currentGlobalNext[3]);
644
645 glm::vec3 origDir = P_next - P_current;
646 glm::vec3 solvedDir = positions[i + 1] - positions[i];
647
648 if (glm::dot(origDir, origDir) < 1e-6f || glm::dot(solvedDir, solvedDir) < 1e-6f) continue;
649
650 glm::quat rot = rotationBetweenVectors(origDir, solvedDir);
651 float w = glm::clamp(ik.targetWeight, 0.0f, 1.0f);
652 glm::quat blendedRot = glm::slerp(glm::quat(1.0f, 0.0f, 0.0f, 0.0f), rot, w);
653
654 // Rotate the current joint
655 glm::mat3 rotPart = glm::mat3(glm::toMat4(blendedRot)) * glm::mat3(origGlobalCurrent);
656 glm::mat4 newGlobal = glm::mat4(rotPart);
657 newGlobal[3] = glm::vec4(glm::mix(P_current, positions[i], w), 1.0f);
658
659 globalTransforms[idxCurrent] = newGlobal;
660
661 // Update local transform of current joint relative to parent
662 int parentIdx = skeleton.joints[idxCurrent].parentIndex;
663 if (parentIdx != -1) {
664 skeleton.joints[idxCurrent].localTransform = glm::inverse(globalTransforms[parentIdx]) * globalTransforms[idxCurrent];
665 } else {
666 skeleton.joints[idxCurrent].localTransform = globalTransforms[idxCurrent];
667 }
668
669 // Recalculate next joint global transform based on the newly rotated current joint
670 globalTransforms[idxNext] = globalTransforms[idxCurrent] * skeleton.joints[idxNext].localTransform;
671 }
672
673 std::vector<bool> resolved(skeleton.joints.size(), false);
674 for (int idx : chainIndices) {
675 resolved[idx] = true;
676 }
677
678 std::function<void(int)> propagate = [&](int jointIdx) {
679 int parentIdx = skeleton.joints[jointIdx].parentIndex;
680 if (parentIdx != -1 && resolved[parentIdx] && !resolved[jointIdx]) {
681 globalTransforms[jointIdx] = globalTransforms[parentIdx] * skeleton.joints[jointIdx].localTransform;
682 resolved[jointIdx] = true;
683 }
684 for (size_t i = 0; i < skeleton.joints.size(); ++i) {
685 if (skeleton.joints[i].parentIndex == jointIdx) {
686 propagate(static_cast<int>(i));
687 }
688 }
689 };
690
691 for (int idx : chainIndices) {
692 propagate(idx);
693 }
694 }

699 void updateEntityAnimation(Entity entity, SkeletonComponent& skeleton, AnimatorComponent& animator, float dt) {
700 if (skeleton.joints.empty()) return;
701
702 // 1. Initialize temporary pose with bind-pose default TRS values
703 std::vector<JointPose> finalPose(skeleton.joints.size());
704 for (size_t i = 0; i < skeleton.joints.size(); ++i) {
705 finalPose[i] = { skeleton.joints[i].bindTranslation, skeleton.joints[i].bindRotation, skeleton.joints[i].bindScale };
706 }
707
708 // 2. Sample poses according to State Machine or fallback to Single Clip Animator
709 if (auto* controller = registry.get<AnimationControllerComponent>(entity)) {
710 if (!controller->currentState.empty()) {
711 // Sample current state
712 const AnimationState* currState = nullptr;
713 for (const auto& state : controller->states) {
714 if (state.name == controller->currentState) {
715 currState = &state;
716 break;
717 }
718 }
719
720 if (currState) {
721 evaluateStatePose(*controller, *currState, controller->currentStateTime, animator, skeleton, finalPose);
722 }
723
724 // Handle crossfading transition
725 if (controller->isCrossfading && !controller->fromState.empty()) {
726 const AnimationState* fState = nullptr;
727 for (const auto& state : controller->states) {
728 if (state.name == controller->fromState) {
729 fState = &state;
730 break;
731 }
732 }
733
734 if (fState) {
735 std::vector<JointPose> fromPose(skeleton.joints.size());
736 for (size_t i = 0; i < skeleton.joints.size(); ++i) {
737 fromPose[i] = { skeleton.joints[i].bindTranslation, skeleton.joints[i].bindRotation, skeleton.joints[i].bindScale };
738 }
739 evaluateStatePose(*controller, *fState, controller->fromStateTime, animator, skeleton, fromPose);
740
741 float weight = glm::clamp(controller->crossfadeProgress / controller->crossfadeDuration, 0.0f, 1.0f);
742 blendPoses(fromPose, finalPose, weight, finalPose);
743 }
744 }
745 }
746 } else {
747 // Fallback to single clip playback
748 AnimationClip* clip = nullptr;
749 if (!animator.animations.empty() &&
750 animator.activeAnimationIndex >= 0 &&
751 animator.activeAnimationIndex < static_cast<int>(animator.animations.size())) {
752 clip = &animator.animations[animator.activeAnimationIndex];
753 }
754
755 if (clip) {
756 if (editorMode.isPlaying) {
757 static int lastPrintedIndex = -999;
758 if (animator.activeAnimationIndex != lastPrintedIndex) {
759 std::cout << "[AnimationSystem] Entity plays animation index: " << animator.activeAnimationIndex
760 << " (Name: " << clip->name << ")" << std::endl;
761 lastPrintedIndex = animator.activeAnimationIndex;
762 }
763 }
764 float speed = editorMode.isPlaying ? (animator.playbackSpeed > 0.0f ? animator.playbackSpeed : 1.0f) : animator.playbackSpeed;
765 animator.currentTime += dt * speed;
766 if (animator.loop) {
767 if (clip->duration > 0.0f) {
768 animator.currentTime = std::fmod(animator.currentTime, clip->duration);
769 } else {
770 animator.currentTime = 0.0f;
771 }
772 } else {
773 animator.currentTime = std::min(animator.currentTime, clip->duration);
774 }
775
776 sampleClip(*clip, animator.currentTime, skeleton, finalPose);
777 }
778 }
779
780 // 3. Write blended pose back to localTransforms
781 for (size_t i = 0; i < skeleton.joints.size(); ++i) {
782 glm::mat4 tMat = glm::translate(glm::mat4(1.0f), finalPose[i].translation);
783 glm::mat4 rMat = glm::toMat4(finalPose[i].rotation);
784 glm::mat4 sMat = glm::scale(glm::mat4(1.0f), finalPose[i].scale);
785 skeleton.joints[i].localTransform = tMat * rMat * sMat;
786 }
787
788 // 4. Resolve Hierarchical Global Transforms (Forward Kinematics)
789 std::vector<glm::mat4> globalTransforms(skeleton.joints.size(), glm::mat4(1.0f));
790 std::vector<bool> resolved(skeleton.joints.size(), false);
791
792 auto resolveJointGlobal = [&](auto& self, int jointIdx) -> glm::mat4 {
793 if (jointIdx == -1) return glm::mat4(1.0f);
794 if (resolved[jointIdx]) return globalTransforms[jointIdx];
795
796 glm::mat4 parentGlobal = self(self, skeleton.joints[jointIdx].parentIndex);
797 globalTransforms[jointIdx] = parentGlobal * skeleton.joints[jointIdx].localTransform;
798 resolved[jointIdx] = true;
799 return globalTransforms[jointIdx];
800 };
801
802 for (size_t i = 0; i < skeleton.joints.size(); ++i) {
803 resolveJointGlobal(resolveJointGlobal, static_cast<int>(i));
804 }
805
806 // 5. Post-FK: Apply IK solver if present and enabled
807 if (auto* ik = registry.get<IKSolverComponent>(entity)) {
808 if (ik->enabled) {
809 if (ik->solverType == IKSolverType::TwoBone) {
810 solve2BoneIK(skeleton, *ik, globalTransforms);
811 } else if (ik->solverType == IKSolverType::FABRIK) {
812 solveFABRIK(skeleton, *ik, globalTransforms);
813 }
814 }
815 }
816
817 // 6. Generate final joint offset matrix palette
818 skeleton.jointMatrices.assign(256, glm::mat4(1.0f));
819 for (size_t i = 0; i < std::min(skeleton.joints.size(), size_t(256)); ++i) {
820 skeleton.jointMatrices[i] = globalTransforms[i] * skeleton.joints[i].inverseBindMatrix;
821 }
822
823 // 7. Upload Joint Matrices Palette to GPU Buffer
824 VkDeviceSize bufferSize = 256 * sizeof(glm::mat4);
825
826 if (!skeleton.gpuBuffer) {
827 skeleton.gpuBuffer = std::make_shared<VulkanBuffer>();
828 skeleton.gpuBuffer->create(
829 renderer.device.getDevice(),
830 renderer.device.getPhysicalDevice(),
831 bufferSize,
832 VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT,
833 VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT | VK_MEMORY_PROPERTY_HOST_COHERENT_BIT
834 );
835
836 renderer.descriptors.allocateJointsDescriptorSet(
837 skeleton.descriptorSet,
838 skeleton.gpuBuffer->get(),
839 bufferSize
840 );
841 }
842
843 skeleton.gpuBuffer->uploadData(skeleton.jointMatrices.data(), bufferSize);
844 sampleAndApplyProperties(entity, animator);
845 }
