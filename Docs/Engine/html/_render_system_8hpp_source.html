156 glm::mat4 getWorldMatrix(Entity entity, int depth = 0) {
157 if (depth > 100) return glm::mat4(1.0f); // Safety depth limit to prevent infinite recursion
158 glm::mat4 model = glm::mat4(1.0f);
159 if (auto* transform = registry.get<Transform>(entity)) {
160 model = transform->matrix();
161 }
162 if (auto* hierarchy = registry.get<HierarchyComponent>(entity)) {
163 if (hierarchy->parent.getId() != Entity::INVALID_ENTITY && registry.isValid(hierarchy->parent)) {
164 // If this is a child mesh entity that is rigid/non-skinned, attach it to its parent bone if matching joint exists
165 if (auto* mesh = registry.get<Mesh>(entity)) {
166 if (!mesh->isSkinned) {
167 std::string targetBone = mesh->parentBoneName.empty() ? mesh->nodeName : mesh->parentBoneName;
168 if (!targetBone.empty()) {
169 if (auto* parentSkeleton = registry.get<SkeletonComponent>(hierarchy->parent)) {
170 int jointIdx = -1;
171
172 // 1. Exact match
173 for (size_t i = 0; i < parentSkeleton->joints.size(); ++i) {
174 if (parentSkeleton->joints[i].name == targetBone) {
175 jointIdx = static_cast<int>(i);
176 break;
177 }
178 }
179
180 // 2. Case-insensitive match fallback
181 if (jointIdx == -1) {
182 auto toLower = [](std::string s) {
183 for (char& c : s) c = static_cast<char>(std::tolower(c));
184 return s;
185 };
186 std::string lowerTarget = toLower(targetBone);
187 for (size_t i = 0; i < parentSkeleton->joints.size(); ++i) {
188 if (toLower(parentSkeleton->joints[i].name) == lowerTarget) {
189 jointIdx = static_cast<int>(i);
190 break;
191 }
192 }
193 }
194
195 // 3. Substring match fallback (handles prefixes/suffixes)
196 if (jointIdx == -1) {
197 for (size_t i = 0; i < parentSkeleton->joints.size(); ++i) {
198 if (parentSkeleton->joints[i].name.find(targetBone) != std::string::npos ||
199 targetBone.find(parentSkeleton->joints[i].name) != std::string::npos) {
200 jointIdx = static_cast<int>(i);
201 break;
202 }
203 }
204 }
205
206 if (jointIdx != -1 && jointIdx < static_cast<int>(parentSkeleton->jointMatrices.size())) {
207 return getWorldMatrix(hierarchy->parent, depth + 1) * parentSkeleton->jointMatrices[jointIdx];
208 }
209 }
210 }
211 }
212 }
213 model = getWorldMatrix(hierarchy->parent, depth + 1) * model;
214 }
215 }
216 return model;
217 }

270 void drawBatches() {
271 VkCommandBuffer cmd = renderer.getCurrentCommandBuffer();
272 VkDescriptorSet cameraSet = renderer.getCameraDescriptorSet();
273
274 // --- Group instances by Mesh + Material using the *existing* instance indices and track their Entities
275 std::unordered_map<std::pair<Mesh*, Material*>, std::vector<std::pair<Entity, size_t>>, pair_hash> batches;
276
277 for (Entity e : entities) {
278 auto* mesh = registry.get<Mesh>(e);
279 auto* mat = registry.get<Material>(e);
280 auto* transform = registry.get<Transform>(e);
281 if (!mesh || !mat || !transform) continue;
282
283 auto it = entityToInstanceIndex.find(e);
284 if (it == entityToInstanceIndex.end()) continue;
285 size_t idx = it->second;
286 batches[{mesh, mat}].push_back({e, idx});
287 }
288
289 for (auto& [key, batch] : batches) {
290 Mesh* mesh = key.first;
291 Material* mat = key.second;
292
293 if (!mesh || !mat || batch.empty() || mesh->vertexBuffer == VK_NULL_HANDLE || mesh->indexBuffer == VK_NULL_HANDLE || mat->pipeline == VK_NULL_HANDLE) continue;
294
295
296 // --- Bind pipeline
297 vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, mat->pipeline);
298
299 // --- Bind descriptor sets (camera at set 0, material sampler at set 1)
300 VkDescriptorSet descriptorSets[2] = {
301 cameraSet,
302 mat->descriptorSet != VK_NULL_HANDLE ? mat->descriptorSet : renderer.getDefaultTextureSet()
303 };
304
305 vkCmdBindDescriptorSets(cmd,
306 VK_PIPELINE_BIND_POINT_GRAPHICS,
307 mat->pipelineLayout,
308 0,
309 2,
310 descriptorSets,
311 0,
312 nullptr);
313
314 // --- Bind vertex buffers
315 VkBuffer vertexBuffers[] = { mesh->vertexBuffer };
316 VkDeviceSize vertexOffsets[] = { 0 };
317 vkCmdBindVertexBuffers(cmd, 0, 1, vertexBuffers, vertexOffsets);
318
319 // If you still have an instance buffer bound elsewhere, it's harmless,
320 // but we don't rely on it here because shader uses push.model.
321 // vkCmdBindVertexBuffers(cmd, 1, 1, &renderer.instanceBuffer.get(), offsets);
322
323 // --- Bind index buffer
324 vkCmdBindIndexBuffer(cmd, mesh->indexBuffer, 0, VK_INDEX_TYPE_UINT32);
325
326 // Draw each instance in this batch individually, pushing its model/color
327 for (auto& [e, instanceIdx] : batch) {
328 // Bind Joint Descriptor Set (Set 2) if entity (or parent) has a Skeleton component
329 SkeletonComponent* skeleton = registry.get<SkeletonComponent>(e);
330 if (!skeleton) {
331 if (auto* hierarchy = registry.get<HierarchyComponent>(e)) {
332 if (hierarchy->parent.getId() != Entity::INVALID_ENTITY && registry.isValid(hierarchy->parent)) {
333 skeleton = registry.get<SkeletonComponent>(hierarchy->parent);
334 }
335 }
336 }
337
338 if (skeleton && skeleton->descriptorSet != VK_NULL_HANDLE) {
339 vkCmdBindDescriptorSets(cmd,
340 VK_PIPELINE_BIND_POINT_GRAPHICS,
341 mat->pipelineLayout,
342 2, // Set 2
343 1,
344 &skeleton->descriptorSet,
345 0,
346 nullptr);
347 }
348
349 // read instance data from renderer.instanceDataCPU
350 const InstanceDataGPU& inst = renderer.instanceDataCPU.get(instanceIdx);
351
352 PushConstants pc{};
353 pc.model = inst.model;
354 pc.color = inst.color;
355 pc.camPos = glm::vec4(getCameraPosition(), 1.0f);
356 if (renderer.hasActiveCamera()) {
357 pc.viewProj = renderer.getActiveCameraViewProj();
358 } else {
359 pc.viewProj = glm::mat4(1.0f);
360 }
361 pc.scale = mat ? mat->roughness : 0.5f;
362 pc.fade = mat ? mat->metallic : 0.0f;
363
364 vkCmdPushConstants(cmd,
365 mat->pipelineLayout,
366 VK_SHADER_STAGE_VERTEX_BIT | VK_SHADER_STAGE_FRAGMENT_BIT,
367 0,

369 &pc);

385 void drawGrids() {
386 VkCommandBuffer cmd = renderer.getCurrentCommandBuffer();
387 VkDescriptorSet cameraSet = renderer.getCameraDescriptorSet();
388
389 for (Entity e : entities) {
390 auto* grid = registry.get<Grid>(e);
391 auto* mat = registry.get<Material>(e);
392 if (!grid || !mat || mat->pipeline == VK_NULL_HANDLE) continue;
393
394
395 // Bind grid pipeline
396 vkCmdBindPipeline(cmd, VK_PIPELINE_BIND_POINT_GRAPHICS, mat->pipeline);
397
398 // Bind grid quad vertex buffer to satisfy pipeline input requirements
399 if (auto* mesh = registry.get<Mesh>(e)) {
400 if (mesh->vertexBuffer != VK_NULL_HANDLE) {
401 VkBuffer vertexBuffers[] = { mesh->vertexBuffer };
402 VkDeviceSize vertexOffsets[] = { 0 };
403 vkCmdBindVertexBuffers(cmd, 0, 1, vertexBuffers, vertexOffsets);
404 }
405 }
406
407 // Bind descriptor sets (camera)
408 vkCmdBindDescriptorSets(cmd,
409 VK_PIPELINE_BIND_POINT_GRAPHICS,
410 mat->pipelineLayout,
411 0,
412 1,
413 &cameraSet,
414 0,
415 nullptr);
416
417 // Get camera position for grid positioning
418 glm::vec3 camPos = getCameraPosition();
419
420 // Push constants for grid shader
421 struct GridPushConstants {
422 glm::mat4 viewProj;
423 glm::vec4 color;
424 glm::vec4 camPos;
425 float scale;
426 float fade;
427 } pc;
428
429 // Use the camera's view-projection matrix
430 if (renderer.hasActiveCamera()) {
431 pc.viewProj = renderer.getActiveCameraViewProj();
432 } else {
433 pc.viewProj = glm::mat4(1.0f);
434 }
435
436 pc.color = grid->color;
437 pc.camPos = glm::vec4(camPos, 1.0f);
438 pc.scale = grid->spacing;
439 pc.fade = grid->size;
440
441 vkCmdPushConstants(cmd,
442 mat->pipelineLayout,
443 VK_SHADER_STAGE_VERTEX_BIT | VK_SHADER_STAGE_FRAGMENT_BIT,
444 0,
445 sizeof(GridPushConstants),
446 &pc);
447
448 // Draw 6 vertices so the shader can emit a full-screen quad as two triangles.
449 vkCmdDraw(cmd, 6, 1, 0, 0);
450 }
